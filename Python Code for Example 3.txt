"""
FAST Fractional-Order PINN for Sharma-Tasso-Olver (STO) Equation
=================================================================
Example 3 from Malagi et al., IJOCTA 2023

PDE:  D_t^alpha u + a*u_x^3 + (3/2)*a*u_xx^2 + a*u_xxx = 0

NOTE: The paper writes this as:
    D_t^alpha u + a*u^3_x + (3/2)*a*u^2_xx + a*u_xxx = 0
which means:
    D_t^alpha u + a*(u_x)^3 + (3/2)*a*(u_xx)^2 + a*u_xxx = 0

Initial condition:
    u(x,0) = sqrt(1/a) * tanh(sqrt(1/a) * x)

Exact solution (alpha=1):
    u(x,t) = sqrt(1/a) * tanh(sqrt(1/a) * (x - t))

q-HAETM series (alpha < 1), from paper eq. 42:
    u0 = sqrt(1/a) * tanh(sqrt(1/a) * x)
    u1 = h*t^alpha * (a*cosh(4*sqrt(1/a)*x) - 2*(a-3)*cosh(2*sqrt(1/a)*x)
                      - 3a - 4) * sech^6(sqrt(1/a)*x)
         / (2*a^2 * Gamma(alpha+1))

Parameters: a=4, n=1, h=-1.858 (optimal h from paper Fig 9)
"""

import torch
import torch.nn as nn
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from scipy.special import gamma as Gamma
import os, warnings
warnings.filterwarnings('ignore')

# ── params (Example 3) ────────────────────────────────────────────
A       = 4.0      # paper uses a=4 for Example 3
N_HAM   = 1.0      # n
H       = -1.858   # optimal hbar from paper Fig 9

ALPHA_LIST = [1.0, 0.75, 0.5]

X_MIN, X_MAX = -5.0, 5.0
T_MIN, T_MAX =  0.0, 0.01   # small t so series stays valid

EPOCHS  = 5000
LR      = 1e-3
N_PDE   = 256
N_IC    = 128
N_BC    = 64
HIDDEN  = 60
LAYERS  = 4
W_IC    = 10.0
W_BC    = 10.0
M_L1    = 4

SAVE_DIR = 'fractional_STO_example3'
os.makedirs(SAVE_DIR, exist_ok=True)

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f"Device: {device}")
torch.manual_seed(42); np.random.seed(42)


# ── helper: sqrt(1/a) ─────────────────────────────────────────────
SA = np.sqrt(1.0 / A)   # sqrt(1/a)


# ── reference / exact solutions ───────────────────────────────────
def ref_sol(x_np, t_np, alpha):
    """
    Reference solution for Example 3.
    alpha=1  -> exact travelling-wave solution
    alpha<1  -> q-HAETM 2-term series (paper eq. 42)
    """
    if alpha == 1.0:
        arg = SA * (x_np - t_np)
        return SA * np.tanh(arg)

    # q-HAETM series: u0 + u1*(1/n)
    # u0
    u0 = SA * np.tanh(SA * x_np)

    # u1 numerator terms (paper eq. 42 top part)
    xi   = SA * x_np
    cosh2 = np.cosh(2*xi)
    cosh4 = np.cosh(4*xi)
    sech6 = (1.0 / np.cosh(xi))**6

    numer = (A * cosh4
             - 2*(A - 3)*cosh2
             - 3*A - 4) * sech6

    u1 = H * t_np**alpha * numer / (2 * A**2 * Gamma(alpha + 1))

    return u0 + u1 * (1.0 / N_HAM)

def ref_sol_safe(x_np, t_np, alpha):
    v = ref_sol(x_np, t_np, alpha)
    if v is None or np.isnan(v) or np.isinf(v):
        return 0.0
    return float(np.clip(v, -10, 10))

def ic_np(x_np):
    """u(x,0) = sqrt(1/a)*tanh(sqrt(1/a)*x)"""
    return SA * np.tanh(SA * x_np)


# ── PINN model ────────────────────────────────────────────────────
class PINN(nn.Module):
    def __init__(self):
        super().__init__()
        lyrs = [nn.Linear(2, HIDDEN), nn.Tanh()]
        for _ in range(LAYERS - 1):
            lyrs += [nn.Linear(HIDDEN, HIDDEN), nn.Tanh()]
        lyrs.append(nn.Linear(HIDDEN, 1))
        self.net = nn.Sequential(*lyrs)
        for m in self.net:
            if isinstance(m, nn.Linear):
                nn.init.xavier_uniform_(m.weight,
                    gain=nn.init.calculate_gain('tanh'))
                nn.init.zeros_(m.bias)

    def forward(self, x, t):
        return self.net(torch.cat([x, t], dim=1))


# ── vectorized L1 Caputo ──────────────────────────────────────────
def caputo_l1_vec(model, x, t, alpha, M=M_L1):
    if alpha == 1.0:
        t_ = t.detach().clone().requires_grad_(True)
        u  = model(x, t_)
        return torch.autograd.grad(u, t_, torch.ones_like(u),
                                   create_graph=True)[0]

    B  = x.shape[0]
    G  = Gamma(2.0 - alpha)
    dt = t / M                                              # (B,1)
    idx    = torch.arange(M+1, dtype=torch.float32,
                          device=device).unsqueeze(0)       # (1,M+1)
    nodes  = idx * dt                                       # (B,M+1)
    x_rep  = x.repeat_interleave(M+1, dim=0)               # (B*(M+1),1)
    t_flat = nodes.reshape(-1, 1)                           # (B*(M+1),1)
    u_flat = model(x_rep, t_flat)                           # (B*(M+1),1)
    u_nodes= u_flat.reshape(B, M+1)                         # (B,M+1)
    j   = torch.arange(M, dtype=torch.float32, device=device)
    bj  = (j+1)**(1-alpha) - j**(1-alpha)
    du  = u_nodes[:, 1:] - u_nodes[:, :-1]
    cap = (bj.unsqueeze(0) * du).sum(dim=1)
    cap = cap / (dt.squeeze(1)**alpha * G)
    return cap.unsqueeze(1)


# ── PDE residual (Example 3 equation) ────────────────────────────
def pde_res(model, x, t, alpha):
    """
    D_t^alpha u + a*(u_x)^3 + (3/2)*a*(u_xx)^2 + a*u_xxx = 0
    """
    x_ = x.detach().clone().requires_grad_(True)
    u   = model(x_, t)
    ux  = torch.autograd.grad(u,   x_, torch.ones_like(u),
                               create_graph=True)[0]
    uxx = torch.autograd.grad(ux,  x_, torch.ones_like(ux),
                               create_graph=True)[0]
    uxxx= torch.autograd.grad(uxx, x_, torch.ones_like(uxx),
                               create_graph=True)[0]
    Dtu = caputo_l1_vec(model, x_, t, alpha)

    # Example 3 PDE: D_t^alpha u + a*u_x^3 + (3/2)*a*u_xx^2 + a*u_xxx = 0
    return Dtu + A*ux**3 + 1.5*A*uxx**2 + A*uxxx


# ── loss ──────────────────────────────────────────────────────────
def loss_fn(model, x_pde, t_pde, x_ic, x_bc, t_bc, alpha):
    # PDE
    res   = pde_res(model, x_pde, t_pde, alpha)
    l_pde = res.pow(2).mean()

    # IC
    t_ic   = torch.zeros_like(x_ic)
    u_ic_p = model(x_ic, t_ic)
    xic_np = x_ic.detach().cpu().numpy().flatten()
    u_ic_r = torch.tensor(
        np.array([ic_np(xi) for xi in xic_np], dtype=np.float32),
        device=device).view(-1, 1)
    l_ic   = (u_ic_p - u_ic_r).pow(2).mean()

    # BC
    xbc_np = x_bc.detach().cpu().numpy().flatten()
    tbc_np = t_bc.detach().cpu().numpy().flatten()
    u_bc_r = torch.tensor(
        np.array([ref_sol_safe(xbc_np[i], tbc_np[i], alpha)
                  for i in range(len(xbc_np))], dtype=np.float32),
        device=device).view(-1, 1)
    u_bc_p = model(x_bc, t_bc)
    l_bc   = (u_bc_p - u_bc_r).pow(2).mean()

    total  = l_pde + W_IC*l_ic + W_BC*l_bc
    return total, l_pde.item(), l_ic.item(), l_bc.item()


# ── training ──────────────────────────────────────────────────────
def train(alpha):
    print(f"\n{'─'*55}")
    print(f"  Training  alpha = {alpha}  ({EPOCHS} epochs)")
    print(f"{'─'*55}")
    model = PINN().to(device)
    opt   = torch.optim.Adam(model.parameters(), lr=LR)
    sch   = torch.optim.lr_scheduler.StepLR(opt, step_size=1500, gamma=0.8)
    hist  = dict(total=[], pde=[], ic=[], bc=[])

    for ep in range(EPOCHS):
        x_pde = (torch.rand(N_PDE,1,device=device)*(X_MAX-X_MIN)+X_MIN)
        t_pde = (torch.rand(N_PDE,1,device=device)*(T_MAX-T_MIN)+T_MIN)
        x_ic  = (torch.rand(N_IC, 1,device=device)*(X_MAX-X_MIN)+X_MIN)
        xL    = torch.full((N_BC,1), X_MIN, device=device)
        xR    = torch.full((N_BC,1), X_MAX, device=device)
        tBC   = (torch.rand(2*N_BC,1,device=device)*(T_MAX-T_MIN)+T_MIN)
        x_bc  = torch.cat([xL, xR])

        loss, lp, li, lb = loss_fn(model, x_pde, t_pde,
                                    x_ic, x_bc, tBC, alpha)
        opt.zero_grad(); loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        opt.step(); sch.step()

        hist['total'].append(loss.item())
        hist['pde'].append(lp)
        hist['ic'].append(li)
        hist['bc'].append(lb)

        if (ep+1) % 1000 == 0:
            print(f"  ep {ep+1:5d} | total {loss.item():.3e} "
                  f"pde {lp:.3e} ic {li:.3e} bc {lb:.3e}")

    model.eval()
    return model, hist


# ── grid helpers ──────────────────────────────────────────────────
@torch.no_grad()
def pred_grid(model, nx=80, nt=40):
    xv = torch.linspace(X_MIN, X_MAX, nx, device=device).float()
    tv = torch.linspace(T_MIN, T_MAX, nt, device=device).float()
    X, T = torch.meshgrid(xv, tv, indexing='ij')
    U = model(X.reshape(-1,1), T.reshape(-1,1)).reshape(nx, nt)
    return X.cpu().numpy(), T.cpu().numpy(), U.cpu().numpy()

def ref_grid(alpha, nx=80, nt=40):
    xv = np.linspace(X_MIN, X_MAX, nx)
    tv = np.linspace(T_MIN, T_MAX, nt)
    X, T = np.meshgrid(xv, tv, indexing='ij')
    U = np.vectorize(lambda xi,ti: ref_sol_safe(xi,ti,alpha))(X, T)
    return X, T, U

# ── TABLE — Absolute errors (paper Table 4 style) ─────────────────
def build_tables(models_dict):
    """
    Paper Table 4: x in {1,2,3,4,5}, t in {0.001, 0.002},
    alpha in {1, 0.75, 0.5}
    """
    x_vals  = [1, 2, 3, 4, 5]
    t_cases = {'t=0.001': 0.001, 't=0.002': 0.002}

    frames = {}
    for t_label, t_val in t_cases.items():
        rows = []
        for xv in x_vals:
            row = {'x': xv}
            for alpha in ALPHA_LIST:
                xT = torch.tensor([[float(xv)]], dtype=torch.float32, device=device)
                tT = torch.tensor([[t_val]],     dtype=torch.float32, device=device)
                with torch.no_grad():
                    u_p = models_dict[alpha](xT, tT).item()
                u_r = ref_sol_safe(float(xv), t_val, alpha)
                row[f'PINN_a{alpha}'] = u_p
                row[f'Ref_a{alpha}']  = u_r
                row[f'Err_a{alpha}']  = abs(u_p - u_r)
            rows.append(row)
        frames[t_label] = pd.DataFrame(rows)

    # Excel
    xls = f'{SAVE_DIR}/error_tables.xlsx'
    with pd.ExcelWriter(xls, engine='openpyxl') as writer:
        for sheet, df in frames.items():
            df.to_excel(writer, sheet_name=sheet, index=False)
    print(f"  ✓ error_tables.xlsx")

    # Print console
    for t_label, df in frames.items():
        print(f"\n{'='*90}")
        print(f"  Absolute Error Table  —  {t_label}  (Example 3)")
        print(f"{'='*90}")
        err_cols = [c for c in df.columns if 'Err' in c]
        rename   = {c: f'|Err| α={c.split("_a")[1]}' for c in err_cols}
        print(df[['x']+err_cols].rename(columns=rename).to_string(index=False))

    return frames

# ── PLOT 8 — PDE Residual Analysis (per alpha) ───────────────────
def plot_residual_analysis(models_dict):
    n_pts = 800
    for alpha in ALPHA_LIST:
        model = models_dict[alpha]
        x_test = (torch.rand(n_pts, 1, device=device) *
                  (X_MAX - X_MIN) + X_MIN).float()
        t_test = (torch.rand(n_pts, 1, device=device) *
                  (T_MAX - T_MIN) + T_MIN).float()

        res_list = []
        bs = 100
        for i in range(0, n_pts, bs):
            xb = x_test[i:i + bs]
            tb = t_test[i:i + bs]
            r = pde_res(model, xb, tb, alpha)
            res_list.append(r.detach().cpu().numpy().flatten())
        residuals = np.concatenate(res_list)

        x_np = x_test.detach().cpu().numpy().flatten()
        t_np = t_test.detach().cpu().numpy().flatten()

        fig, axes = plt.subplots(1, 2, figsize=(13, 5))

        # histogram
        ax = axes[0]
        ax.hist(residuals, bins=50, color='#7b2d8b',
                alpha=0.8, edgecolor='white')
        mean_r = np.mean(residuals)
        ax.axvline(mean_r, color='red', ls='--', lw=2,
                   label=f'Mean: {mean_r:.2e}')
        ax.set_xlabel('PDE Residual', fontsize=12)
        ax.set_ylabel('Frequency', fontsize=12)
        ax.set_title('PDE Residual Distribution', fontsize=13)
        ax.legend();
        ax.grid(True, alpha=0.3)

        # scatter |residual| over domain

# ── PLOT 10 — Weights & Biases Distribution (per alpha) ──────────
def plot_weights_distribution(models_dict):
    for alpha in ALPHA_LIST:
        model = models_dict[alpha]
        linear_layers = [m for m in model.net if isinstance(m, nn.Linear)]

        params = []
        for idx, layer in enumerate(linear_layers):
            params.append((f'layer_{idx}_weights',
                           layer.weight.data.cpu().numpy()))
            params.append((f'layer_{idx}_biases',
                           layer.bias.data.cpu().numpy()))

        n_params = len(params)
        n_cols = 4
        n_rows = (n_params + n_cols - 1) // n_cols

        fig, axes = plt.subplots(n_rows, n_cols,
                                 figsize=(4.5 * n_cols, 3.5 * n_rows))
        axes = axes.flatten()

        for idx, (name, arr) in enumerate(params):
            ax = axes[idx]
            color = 'blue' if 'weights' in name else 'green'
            ax.hist(arr.flatten(), bins=40,
                    alpha=0.75, color=color, edgecolor='white')
            mean_v = np.mean(arr)
            ax.axvline(mean_v, color='red', ls='--', lw=1.2)
            ax.set_title(f'{name}\nShape: {arr.shape}', fontsize=8)
            lbl = 'Weight Values' if 'weights' in name else 'Bias Values'
            ax.set_xlabel(lbl, fontsize=7)
            ax.set_ylabel('Frequency', fontsize=7)
            ax.text(0.04, 0.95, f'Mean: {mean_v:.3e}',
                    transform=ax.transAxes, fontsize=7,
                    verticalalignment='top')
            ax.grid(True, alpha=0.25)

        for idx in range(n_params, len(axes)):
            axes[idx].axis('off')

        plt.suptitle(f'Weights and Biases Distribution  —  α={alpha} ,',
                     fontsize=13, y=1.01)
        plt.tight_layout()
        tag = str(alpha).replace('.', 'p')
        plt.savefig(f'{SAVE_DIR}/10_weights_dist_alpha{tag}.png',
                    dpi=150, bbox_inches='tight')
        plt.close()
    print("  ✓ 10_weights_dist_alpha*.png")


# ── NEW FUNCTION: Save all solutions to Excel ─────────────────────
def save_all_solutions_to_excel(models_dict):
    """
    Save complete solution grids (all x,t points) to Excel files for each alpha.
    Each Excel file will contain PINN predictions, Reference solutions, and Errors.
    """
    nx, nt = 80, 40  # Same grid resolution as plotting

    for alpha in ALPHA_LIST:
        print(f"  Saving complete solution for α={alpha}...")

        # Get grid predictions
        X, T, Up_pred = pred_grid(models_dict[alpha], nx=nx, nt=nt)
        _, _, Ur_ref = ref_grid(alpha, nx=nx, nt=nt)

        # Flatten the grids for Excel storage
        x_flat = X.flatten()
        t_flat = T.flatten()
        u_pinn_flat = Up_pred.flatten()
        u_ref_flat = Ur_ref.flatten()
        error_flat = np.abs(Up_pred - Ur_ref).flatten()

        # Create DataFrame with all values
        df = pd.DataFrame({
            'x': x_flat,
            't': t_flat,
            'PINN_solution': u_pinn_flat,
            'Reference_solution': u_ref_flat,
            'Absolute_Error': error_flat
        })

        # Create filename
        filename = f'{SAVE_DIR}/complete_solutions_alpha_{str(alpha).replace(".", "p")}.xlsx'

        with pd.ExcelWriter(filename, engine='openpyxl') as writer:
            # Sheet 1: All flattened data (complete dataset)
            df.to_excel(writer, sheet_name='All_Data_Flattened', index=False)

            # Sheet 2: Grid format - X coordinates
            pd.DataFrame(X).to_excel(writer, sheet_name='X_grid', index=False, header=False)

            # Sheet 3: Grid format - T coordinates
            pd.DataFrame(T).to_excel(writer, sheet_name='T_grid', index=False, header=False)

            # Sheet 4: PINN solution in grid format
            pd.DataFrame(Up_pred).to_excel(writer, sheet_name='PINN_solution_grid', index=False, header=False)

            # Sheet 5: Reference solution in grid format
            pd.DataFrame(Ur_ref).to_excel(writer, sheet_name='Reference_solution_grid', index=False, header=False)

            # Sheet 6: Error in grid format
            pd.DataFrame(np.abs(Up_pred - Ur_ref)).to_excel(writer, sheet_name='Error_grid', index=False, header=False)

            # Sheet 7: Summary statistics
            summary = pd.DataFrame({
                'Metric': ['Min', 'Max', 'Mean', 'Std', 'RMSE'],
                'PINN_Solution': [
                    np.min(Up_pred), np.max(Up_pred), np.mean(Up_pred), np.std(Up_pred),
                    np.sqrt(np.mean((Up_pred - Ur_ref)**2))
                ],
                'Reference_Solution': [
                    np.min(Ur_ref), np.max(Ur_ref), np.mean(Ur_ref), np.std(Ur_ref),
                    np.sqrt(np.mean((Up_pred - Ur_ref)**2))
                ],
                'Absolute_Error': [
                    np.min(error_flat), np.max(error_flat), np.mean(error_flat), np.std(error_flat),
                    np.sqrt(np.mean((Up_pred - Ur_ref)**2))
                ]
            })
            summary.to_excel(writer, sheet_name='Summary_Statistics', index=False)

            # Sheet 8: Sample of first 1000 points (for quick preview)
            df.head(1000).to_excel(writer, sheet_name='Sample_First_1000', index=False)

        print(f"  ✓ Saved complete solution grid for α={alpha} ({len(df)} points)")
